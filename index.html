QuEra Bloqade & Quantum SDKs

Python and Julia code snippets ready for local emulation and cloud QPU execution. 

from bloqade import var, cast
from bloqade.atom_arrangement import Square

# 1. Define 2D Neutral Atom Tweezer Geometry (QuEra Aquila Spec)
lattice = Square(4, 4, lattice_constant=5.0)

# 2. Program Analog Time-Dependent Pulse Driving (Rabi Frequency & Detuning)
durations = [0.2, 1.6, 0.2]
rabi_pulse = var("rabi").piecewise_linear(durations, [0.0, 2.5, 2.5, 0.0])
detuning_pulse = var("delta").piecewise_linear(durations, [-2.0, -2.0, 3.0, 3.0])

# 3. Build & Emulate Rydberg Hamiltonian Program
program = lattice.rydberg.rabi.amplitude.uniform.piecewise_linear(
    durations, [0.0, 2.5, 2.5, 0.0]
).detuning.uniform.piecewise_linear(durations, [-2.0, -2.0, 3.0, 3.0])

# 4. Run Statevector Simulation
batch = program.bloqade.python().run(1000)
print("Rydberg Population Results:", batch.report().counts())
